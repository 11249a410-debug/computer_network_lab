def chat_server():
# Create TCP socket
server_socket = socket.socket(
socket.AF_INET,
socket.SOCK_STREAM
)
# Allow immediate reuse of the port
server_socket.setsockopt(socket.SOL_SOCKET,
socket.SO_REUSEADDR,
1
)
# Bind server to localhost and port 55555
server_socket.bind((&#39;localhost&#39;, 55555))# Listen for incoming connections
server_socket.listen(5)
print(&quot;Server listening on port 55555...&quot;)
while True:
# Accept a new client
client_socket, client_address = server_socket.accept()
# Add client to the client list
clients.append(client_socket)
# Create a separate thread for a client 
thread = threading.Thread(
target=handle_client,
args=(client_socket,)
)
# Allow thread to terminate when server stops
thread.daemon = True
# Start client thread
thread.start()
if __name__ == &#39;__main__&#39;:
chat_server()
